SET NAMES utf8mb4;

-- ----------------------------
-- Table structure for area
-- ----------------------------
DROP TABLE IF EXISTS `area`;
CREATE TABLE `area`  (
  `code` varchar(6) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci NOT NULL DEFAULT '' COMMENT '地址编码',
  `name` varchar(50) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci NOT NULL COMMENT '地区名称',
  `full_name` varchar(50) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci NULL DEFAULT NULL COMMENT '地区全称',
  `parent_code` varchar(6) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci NOT NULL COMMENT '上级编码',
  `level` tinyint NOT NULL COMMENT '地区级别：1-省  2-市  3-县',
  PRIMARY KEY (`code`) USING BTREE
) ENGINE = InnoDB CHARACTER SET = utf8mb3 COLLATE = utf8mb3_general_ci COMMENT = '三级地区信息表' ROW_FORMAT = Dynamic;

-- ----------------------------
-- Table structure for browse
-- ----------------------------
DROP TABLE IF EXISTS `browse`;
CREATE TABLE `browse`  (
  `id` int UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '浏览记录编号',
  `who` int UNSIGNED NOT NULL COMMENT '浏览用户编号 (外键)',
  `what` int UNSIGNED NOT NULL COMMENT '浏览通知编号 (外键)',
  `when` timestamp NOT NULL COMMENT '浏览时间',
  PRIMARY KEY (`id`) USING BTREE,
  INDEX `key_browse_who_user_id`(`who` ASC) USING BTREE,
  INDEX `key_browse_what_notice_id`(`what` ASC) USING BTREE,
  CONSTRAINT `key_browse_what_notice_id` FOREIGN KEY (`what`) REFERENCES `notice` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `key_browse_who_user_id` FOREIGN KEY (`who`) REFERENCES `user` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE = InnoDB AUTO_INCREMENT = 2 CHARACTER SET = utf8mb4 COLLATE = utf8mb4_general_ci COMMENT = '通知浏览记录表' ROW_FORMAT = DYNAMIC;

-- Table structure for checklist

  PRIMARY KEY (`id`) USING BTREE,
  INDEX `fk_action_user_user_id`(`action_user` ASC) USING BTREE,
  INDEX `fk_vendor_id_vendor_id`(`vendor_id` ASC) USING BTREE,
  INDEX `fk_verify_user_user_id`(`verify_user` ASC) USING BTREE,
  CONSTRAINT `fk_action_user_user_id` FOREIGN KEY (`action_user`) REFERENCES `user` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fk_vendor_id_vendor_id` FOREIGN KEY (`vendor_id`) REFERENCES `vendor` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fk_verify_user_user_id` FOREIGN KEY (`verify_user`) REFERENCES `user` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE = InnoDB AUTO_INCREMENT = 7 CHARACTER SET = utf8mb4 COLLATE = utf8mb4_general_ci COMMENT = '药品出入库清单表' ROW_FORMAT = Dynamic;

-- Table structure for notice
-- ----------------------------
DROP TABLE IF EXISTS `notice`;
CREATE TABLE `notice`  (
  `id` int UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '通知编号',
  `title` varchar(50) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci NOT NULL COMMENT '通知标题',
  `content` varchar(1000) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci NOT NULL COMMENT '通知内容',
  `sender` int UNSIGNED NOT NULL COMMENT '发布用户 (外键)',
  `when` datetime NOT NULL COMMENT '发布时间',
  `receiver` varchar(50) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci NOT NULL COMMENT '接收用户类型',
  `status` tinyint UNSIGNED NOT NULL COMMENT '通知状态 (1-未发布 2-已发布 3-已销毁)',
  `type` tinyint UNSIGNED NOT NULL COMMENT '通知类型 (1-普通通知 2-维护通知)',
  PRIMARY KEY (`id`) USING BTREE,
  INDEX `notice_sender_user_id`(`sender` ASC) USING BTREE,
  INDEX `notice_receiver_identity_id`(`receiver` ASC) USING BTREE,
  CONSTRAINT `notice_sender_user_id` FOREIGN KEY (`sender`) REFERENCES `user` (`id`) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE = InnoDB AUTO_INCREMENT = 2 CHARACTER SET = utf8mb4 COLLATE = utf8mb4_general_ci COMMENT = '通知信息表' ROW_FORMAT = DYNAMIC;

-- ----------------------------
-- Table structure for record
-- ----------------------------
DROP TABLE IF EXISTS `record`;
CREATE TABLE `record`  (
  `id` int UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '记录编号',
  `when` date NOT NULL COMMENT '记录时间',
  `what` int UNSIGNED NOT NULL COMMENT '药品编号 (外键)',
  `type` tinyint NOT NULL COMMENT '操作类型 (1-入库 2-出库)',
  `count` int UNSIGNED NOT NULL COMMENT '药品数量',
  `price` decimal(10, 2) UNSIGNED NOT NULL COMMENT '药品单价',
  `birthday` date NOT NULL COMMENT '生产日期',
  `from` tinyint UNSIGNED NOT NULL COMMENT '供应商编号 (外键)',
  `who` int UNSIGNED NOT NULL COMMENT '操作用户 (外键)',
  `parent` int UNSIGNED NOT NULL COMMENT '清单编号 (外键)',
  `other` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci NOT NULL COMMENT '备注信息',
  PRIMARY KEY (`id`) USING BTREE,
  INDEX `key_what_medicine_id`(`what` ASC) USING BTREE,
  INDEX `key_who_user_id`(`who` ASC) USING BTREE,
  INDEX `record_from_vendor_id`(`from` ASC) USING BTREE,
  CONSTRAINT `record_from_vendor_id` FOREIGN KEY (`from`) REFERENCES `vendor` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `record_what_medicine_id` FOREIGN KEY (`what`) REFERENCES `medicine` (`id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `record_who_user_id` FOREIGN KEY (`who`) REFERENCES `user` (`id`) ON DELETE CASCADE ON UPDATE CASCADE

spring:

    password: admin123

      port: 6379

  mapper-locations: classpath:mapper/*.xml
  configuration:
    map-underscore-to-camel-case: true    #开启驼峰命名映射
    cache-enabled: true    #开启二级缓存(mapper级别)

server:
  servlet:
    session:
      timeout: 60m #设置会话过期时间
  port: 8080
